       UPPER(last_name) AS upper_name,
       CHAR_LENGTH(last_name) AS chars,
       OCTET_LENGTH(last_name) AS bytes,
       SUM(1) AS one
FROM member
WHERE last_name COLLATE utf8mb4_unicode_ci = 'MUNOZ'
   OR last_name REGEXP '^[ÀÉÎ]'
   OR first_name LIKE '%\_%' ESCAPE '\\'
GROUP BY last_name;
